/**
 * Get the CSRF token from the page meta tag.
 * @returns {string|null}
 */
function getCsrfToken() {
    const meta = document.querySelector('meta[name="csrfToken"]');
    return meta ? meta.getAttribute("content") : null;
}

/**
 * Create the upload handler for TinyMCE images.
 * Returns a Promise-based function compatible with TinyMCE 6+.
 *
 * @param {string} uploadUrl - The endpoint for image uploads
 * @returns {Function}
 */
function createImageUploadHandler(uploadUrl) {
    return function (blobInfo, progress) {
        return new Promise(function (resolve, reject) {
            const xhr = new XMLHttpRequest();
            xhr.open("POST", uploadUrl);
            xhr.withCredentials = true;

            xhr.upload.onprogress = function (e) {
                if (e.lengthComputable) {
                    progress((e.loaded / e.total) * 100);
                }
            };

            xhr.onload = function () {
                if (xhr.status < 200 || xhr.status >= 300) {
                    return reject("HTTP Error: " + xhr.status);
                }

                let json;
                try {
                    json = JSON.parse(xhr.responseText);
                } catch (err) {
                    console.error(
                        "TinyMCE upload invalid JSON response:",
                        xhr.responseText,
                        err,
                    );
                    return reject("Invalid JSON response");
                }

                if (!json.success || !json.image || !json.image.url) {
                    console.error(
                        "TinyMCE upload server response (error path):",
                        json,
                    );
                    return reject(json.error || "Upload failed");
                }

                // Return WebP URL with fallback for older browsers
                const imageUrl = json.image.url;
                const webpUrl = imageUrl.includes("?")
                    ? imageUrl + "&fm=webp"
                    : imageUrl + "?fm=webp";

                resolve(webpUrl);
            };

            xhr.onerror = function () {
                reject("Image upload failed due to network error");
            };

            const formData = new FormData();
            formData.append("upload", blobInfo.blob(), blobInfo.filename());

            const csrf = getCsrfToken();
            if (csrf) {
                xhr.setRequestHeader("X-CSRF-Token", csrf);
            }

            xhr.send(formData);
        });
    };
}

/**
 * Default configuration for TinyMCE with Bootstrap styling.
 *
 * @param {Object} options - Configuration override options
 * @param {string} options.selector - CSS selector for the textarea
 * @param {string} [options.uploadUrl='/admin/images/upload'] - Image upload endpoint
 * @param {boolean} [options.menubar=false] - Show menubar
 * @param {number} [options.minHeight=400] - Minimum editor height
 * @param {Function} [options.onInit] - Callback when editor initializes
 * @param {Function} [options.onSave] - Custom save handler
 * @returns {Object} TinyMCE configuration object
 */
export function createTinyMCEConfig(options = {}) {
    const {
        selector = "#body-editor",
        uploadUrl = "/admin/images/upload",
        menubar = false,
        minHeight = 400,
        onInit = null,
        onSave = null,
    } = options;

    const config = {
        license_key: "gpl",
        selector: selector,
        menubar: menubar,
        min_height: minHeight,
        resize: true,
        statusbar: true,
        branding: false,

        // Plugins for comprehensive editing
        plugins: [
            "image",
            "code",
            "lists",
            "advlist",
            "media",
            "preview",
            "quickbars",
            "save",
            "visualblocks",
            "visualchars",
            "table",
            "link",
            "autolink",
            "searchreplace",
            "fullscreen",
            "wordcount",
            "help",
        ].join(" "),

        // Main toolbar with all needed controls
        toolbar: [
            "undo redo | blocks styles | bold italic underline strikethrough | forecolor backcolor",
            "alignleft aligncenter alignright | bullist numlist | outdent indent | blockquote",
            "link image media table | removeformat visualblocks | code fullscreen preview | help",
        ].join(" | "),

        // Quick selection toolbar
        quickbars_selection_toolbar:
            "bold italic underline | quicklink blockquote | bullist numlist",
        quickbars_insert_toolbar: "quickimage quicktable",

        // Block formats in dropdown
        block_formats:
            "Paragraph=p; Heading 1=h1; Heading 2=h2; Heading 3=h3; Heading 4=h4; Heading 5=h5; Heading 6=h6; Blockquote=blockquote; Pre=pre",

        // Custom formats for styling
        formats: getFormats(),
        style_formats: getStyleFormats(),

        // Content styling (embedded for preview)
        content_style: getContentStyles(),

        // Also use external CSS for full theme support
        content_css: [
            "https://cdn.jsdelivr.net/npm/bootstrap@5.3.2/dist/css/bootstrap.min.css",
            "/css/blog-content.css",
        ],

        // Image handling
        image_title: true,
        automatic_uploads: true,
        images_upload_url: uploadUrl,
        images_upload_credentials: true,
        images_reuse_filename: true,
        convert_urls: false,
        relative_urls: false,

        // Custom upload handler
        images_upload_handler: createImageUploadHandler(uploadUrl),

        // Table configuration
        table_default_styles: {
            width: "100%",
        },
        table_class_list: [
            { title: "Default", value: "" },
            { title: "Striped", value: "table table-striped" },
            { title: "Bordered", value: "table table-bordered" },
            { title: "Hover", value: "table table-hover" },
            { title: "Small", value: "table table-sm" },
        ],
        table_responsive_width: true,

        // Link configuration
        link_assume_external_targets: true,
        link_default_target: "_blank",
        link_class_list: [
            { title: "None", value: "" },
            { title: "Button Primary", value: "btn btn-primary" },
            { title: "Button Secondary", value: "btn btn-secondary" },
            { title: "Button Link", value: "btn btn-link" },
        ],

        // Visual aids
        visualblocks_default_state: false,

        // Valid elements (allow Bootstrap classes and responsive attributes)
        extended_valid_elements:
            "img[class|src|srcset|sizes|alt|title|width|height|loading|style|data-*]," +
            "picture[class|style],source[srcset|sizes|type|media]," +
            "figure[class|style],figcaption[class|style]," +
            "iframe[src|width|height|frameborder|allowfullscreen|class|style|title]," +
            "video[src|controls|autoplay|loop|muted|poster|class|style|width|height]," +
            "table[class|style|border|cellpadding|cellspacing]," +
            "div[class|style|data-*],span[class|style|data-*]",

        // Initialization callback
        setup: function (editor) {
            editor.on("init", function () {
                if (typeof onInit === "function") {
                    onInit(editor);
                }
            });

            // Custom save handler
            if (typeof onSave === "function") {
                editor.on("SaveContent", function (e) {
                    onSave(editor, e);
                });
            }

            // Add custom image insertion command
            editor.addCommand("insertResponsiveImage", function (ui, value) {
                const { imageId, alt = "", classes = "" } = value || {};
                if (!imageId) return;

                const webpUrl = `/images/serve/${imageId}?fm=webp&w=800`;
                const fallbackUrl = `/images/serve/${imageId}?w=800`;

                const html = `
                    <picture>
                        <source srcset="${webpUrl}" type="image/webp">
                        <img src="${fallbackUrl}" alt="${alt}" class="img-fluid ${classes}" loading="lazy">
                    </picture>
                `;
                editor.insertContent(html);
            });
        },
    };

    return config;
}

/**
 * Initialize TinyMCE on a selector with Bootstrap configuration.
 *
 * @param {Object} options - Configuration options
 * @returns {Promise} Resolves when TinyMCE is initialized
 */
export function initTinyMCE(options = {}) {
    return new Promise((resolve, reject) => {
        if (typeof window.tinymce === "undefined") {
            reject(new Error("TinyMCE is not loaded"));
            return;
        }

        const config = createTinyMCEConfig({
            ...options,
            onInit: (editor) => {
                if (typeof options.onInit === "function") {
                    options.onInit(editor);
                }
                resolve(editor);
            },
        });

        window.tinymce.init(config);
    });
}

/**
 * Insert a responsive image with WebP support at cursor position.
 *
 * @param {Object} editor - TinyMCE editor instance
 * @param {number|string} imageId - Image ID to insert
 * @param {Object} options - Image options
 * @param {string} [options.alt=''] - Alt text
 * @param {string} [options.classes=''] - Additional CSS classes
 * @param {string} [options.position='inline'] - Position: 'inline', 'left', 'right', 'center'
 * @param {number} [options.width=800] - Max width for the image
 */
export function insertResponsiveImage(editor, imageId, options = {}) {
    const {
        alt = "",
        classes = "",
        position = "inline",
        width = 800,
    } = options;

    if (!editor || !imageId) return;

    const webpUrl = `/images/serve/${imageId}?fm=webp&w=${width}`;
    const fallbackUrl = `/images/serve/${imageId}?w=${width}`;

    let positionClass = "";
    switch (position) {
        case "left":
            positionClass = "img-float-left";
            break;
        case "right":
            positionClass = "img-float-right";
            break;
        case "center":
            positionClass = "img-center";
            break;
    }

    const allClasses = [positionClass, "img-fluid", classes]
        .filter(Boolean)
        .join(" ");

    const html = `
        <picture class="${position === "inline" ? "" : positionClass}">
            <source srcset="${webpUrl}" type="image/webp">
            <img src="${fallbackUrl}" alt="${alt}" class="${allClasses}" loading="lazy">
        </picture>
    `;

    editor.insertContent(html.trim());
}

export default {
    createTinyMCEConfig,
    initTinyMCE,
    insertResponsiveImage,
    getCsrfToken,
};
